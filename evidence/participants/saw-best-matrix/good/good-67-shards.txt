=== bos-game/run67.sh
#!/bin/sh
# Good-matrix census at n=67, one shard per argument "di:lo:hi" (decomposition
# index, outer-table slices [lo,hi) of GOOD_OUTER_CHUNKS=8).  Builds the engine,
# validates it on the stored n=37 census, then runs the shards in order under a
# memory-capped user scope at nice 19.  Resumable: a shard with a .done marker is
# skipped, and a shard with a checkpoint resumes from it.
#   usage: run67.sh THREADS MEMMAX di:lo:hi [di:lo:hi ...]
set -u
cd "$HOME/gc67" || exit 1
T=$1; MEM=$2; shift 2
L=$HOME/gc67/queue.log
export XDG_RUNTIME_DIR=/run/user/$(id -u) DBUS_SESSION_BUS_ADDRESS=unix:path=/run/user/$(id -u)/bus
export OMP_WAIT_POLICY=PASSIVE
if [ ! -x good_c ] || [ good_c.c -nt good_c ]; then
  gcc -O3 -march=native -fopenmp -DGS_WIDE -DGC_WIDE -o good_c good_c.c -lm 2>>"$L" || { echo "build failed $(date -Is)" >> "$L"; exit 1; }
  rm -f val37.ok
fi
if [ ! -f val37.ok ]; then
  GOOD_OUTER_CHUNKS=4 nice -n 19 ./good_c 37 "$T" val37.txt > val37.log 2>&1
  want=$(grep -c '^[+-]' good-37.ref); got=$(grep -c '^[+-]' val37.txt)
  echo "validation n=37: reference rows=$want local rows=$got $(date -Is)" >> "$L"
  grep '^[+-]' good-37.ref | sort -u > ref.sorted; grep '^[+-]' val37.txt | sort -u > val.sorted
  if [ "$want" = "$got" ] && cmp -s ref.sorted val.sorted; then touch val37.ok; else echo "validation FAILED; n=67 not started" >> "$L"; exit 1; fi
fi
for u in "$@"; do
  di=${u%%:*}; r=${u#*:}; lo=${r%%:*}; hi=${r#*:}
  tag="d${di}-s${lo}-${hi}"
  [ -f "$tag.done" ] && continue
  echo "shard $tag starting, threads $T, MemoryMax $MEM $(date -Is)" >> "$L"
  GOOD_OUTER_CHUNKS=8 GOOD_DI=$di GOOD_OC_LO=$lo GOOD_OC_HI=$hi \
  systemd-run --user --scope -q -p MemoryMax="$MEM" -p MemorySwapMax=0 \
    nice -n 19 ./good_c 67 "$T" "out-$tag.txt" "ckpt-$tag" 600 >> "out-$tag.log" 2>> "err-$tag.log"
  rc=$?
  echo "shard $tag rc=$rc $(date -Is)" >> "$L"
  [ $rc -eq 0 ] && touch "$tag.done" || exit $rc
done
echo "all shards done $(date -Is)" >> "$L"
