# Quadruple with a skew block at n = 131 from the Paley matrix of order 524.
# A literature construction, not a search result and not new. It extends evidence/literature/paley-four-circulant.txt
# (n = 83 and 125) to q = 523 with the same two scripts, copied below unchanged except for the q in the run line.
# For every prime power q = 4n - 1 the source (Xia, Xia, Seberry and Wu, An infinite family of Goethals-Seidel
# arrays, Discrete Appl. Math. 145 (2005) 498-504, doi:10.1016/j.dam.2003.06.007) proves that circulant A, B, C, D
# of order n exist with PAFs summing to zero, A of skew type, B symmetric and C = D. Its rows were not transcribed;
# the rows below were generated on 2026-11-23 from the Paley matrix (Paley, On orthogonal matrices, J. Math. Phys.
# 12 (1933) 311-320, doi:10.1002/sapm1933121311) and have that form.
#
# Construction. As in evidence/literature/paley-four-circulant.txt: q = 4n - 1 prime (q = 3 mod 8),
# GF(q^2) = GF(q)[i] with i^2 = -1, w a generator of GF(q^2)*, chi the quadratic character of GF(q),
# f(d) = chi(Im w^d) for 0 <= d <= q and X_r(k) = (-1)^k f(r + 4k) for r = 0..3 and 0 <= k < n. After sign changes
# of rows and columns the Paley core of order q + 1 is the block matrix of the circulants of X_0..X_3 and the cyclic
# shift. The stored rows are (X_0 with X_0(0) set to 1, X_2 shifted to be symmetric, X_1, X_1). Since q = 523 < 600,
# xia_form.py runs gen.py with check_matrix=True: it builds the full Paley core of order 524, checks zero diagonal,
# skewness and M M^T = q I, and checks the block form entry by entry, then checks the PAF identity, the skew first
# row, the symmetric second row and that X_3 is a shifted reversal of X_1 up to sign.
# Not checked: whether the Goethals-Seidel matrix of the stored rows is equivalent to the Paley matrix.
# Rerunning xia_form.py 331 499 reproduces the n = 83 and 125 rows of paley-four-circulant.txt exactly.
#
# n = 131: q = 523, w = 2 + 1i, shift 65, row sums [1, -19, 9, 9]
#
# gen.py:
#   """Paley core of order q+1 (q prime, q = 3 mod 8) as four circulants of order n=(q+1)/4.
#
#   Points of the projective line over GF(q) are w^d GF(q)*, d = 0..q, for a
#   generator w of GF(q^2)* with GF(q^2) = GF(q)[i], i^2 = -1.  The Paley core is
#   M[a][b] = chi(det(w^a, w^b)).  With f(d) = chi(Im w^d) and
#   X_r[k] = (-1)^k f(r + 4k), this script checks, entry by entry, that after
#   scaling row (r,k) by (-1)^(r+k) and column (r',k') by (-1)^k' the core is
#
#       [  A   B   C   D ]
#       [ -PD  A   B   C ]
#       [ -PC -PD  A   B ]
#       [ -PB -PC -PD  A ]
#
#   with A,B,C,D the circulants of X_0..X_3 and P the cyclic shift, and that
#   (X_0 + delta_0, X_1, X_2, X_3) has PAFs summing to zero with X_0 skew.
#   """
#   import json

#   def isprime(m):
#       return m > 1 and all(m % d for d in range(2, int(m**0.5) + 1))
#
#
#   def quadruple(q, check_matrix=True):

#       n = (q + 1) // 4
#       qr = {x * x % q for x in range(1, q)}
#       chi = lambda a: 0 if a % q == 0 else (1 if a % q in qr else -1)
#       mul = lambda x, y: ((x[0] * y[0] - x[1] * y[1]) % q, (x[0] * y[1] + x[1] * y[0]) % q)
#       # generator of GF(q^2)*: brute force on the order

#           (a, b)

#           if all(power((a, b), N // p) != (1, 0) for p in primes)
#       )
#       pts = [(1, 0)]
#       for _ in range(q):
#           pts.append(mul(pts[-1], w))
#       f = [chi(p[1]) for p in pts]
#       X = [[(-1) ** k * f[r + 4 * k] for k in range(n)] for r in range(4)]
#       assert X[0][0] == 0 and sum(v == 0 for x in X for v in x) == 1
#       if check_matrix:
#           det = lambda x, y: (x[0] * y[1] - x[1] * y[0]) % q
#           M = [[chi(det(pts[a], pts[b])) for b in range(q + 1)] for a in range(q + 1)]
#           # Paley core: zero diagonal, skew, M M^T = q I
#           assert all(M[a][a] == 0 for a in range(q + 1))
#           assert all(M[a][b] == -M[b][a] for a in range(q + 1) for b in range(a))
#           pos = [sum(1 << b for b in range(q + 1) if M[a][b] == 1) for a in range(q + 1)]
#           neg = [sum(1 << b for b in range(q + 1) if M[a][b] == -1) for a in range(q + 1)]
#           pc = lambda v: bin(v).count("1")
#           for a in range(q + 1):
#               for b in range(a, q + 1):
#                   dot = pc(pos[a] & pos[b]) + pc(neg[a] & neg[b]) - pc(pos[a] & neg[b]) - pc(neg[a] & pos[b])
#                   assert dot == (q if a == b else 0)
#           # block form
#           for r in range(4):
#               for k in range(n):
#                   a = r + 4 * k
#                   for r2 in range(4):
#                       t, c = (r2 - r) % 4, int(r2 < r)
#                       for k2 in range(n):
#                           b = r2 + 4 * k2
#                           lhs = (-1) ** (r + k + k2) * M[a][b]
#                           assert lhs == (-1) ** c * X[t][(k2 - k - c) % n], (a, b)
#       X[0][0] = 1
#       paf = lambda x, s: sum(x[j] * x[(j + s) % n] for j in range(n))
#       assert all(sum(paf(x, s) for x in X) == 0 for s in range(1, n))
#       assert all(X[0][-j % n] == -X[0][j] for j in range(1, n))
#       return n, w, X
